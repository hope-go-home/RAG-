# 第 8 章：系统编程与系统调用

## 8.1 什么是系统调用？

### 从用户态到内核态

之前讲过用户态和内核态。系统调用就是**用户程序请求内核帮忙的"正规渠道"**。

```
用户程序 (用户态)
    │
    │ 想要：读取文件、创建进程、分配内存
    │ 但：不能直接操作硬件（安全限制）
    │
    ├──→ 系统调用 (System Call)
    │    如：read(), fork(), malloc() [底层用 brk()]
    │
    ↓
内核 (内核态)
    │
    │ 执行：操作硬盘、创建进程表项、管理页表
    │
    ↓
返回结果给用户程序
```

### ⚡ 系统调用的执行过程

```
以 read() 为例：
    ↓
1. 用户程序调用 read(fd, buf, size)
    ↓
2. 进入 C 库（glibc）的 read 包装函数
    ↓
3. 包装函数把参数放到寄存器
    ↓
4. 执行 syscall 指令（或 int 0x80）
    ↓
5. CPU 切换到内核态
    ↓
6. 内核根据系统调用号查找 sys_call_table
    ↓
7. 执行对应的内核函数（如 sys_read）
    ↓
8. 内核操作 VFS（虚拟文件系统）→ 硬盘驱动 → 硬盘
    ↓
9. 数据读到内核缓冲区 → 复制到用户缓冲区
    ↓
10. 返回用户态，read() 返回读取的字节数
```

### 指针 — C 语言的灵魂

```c
#include <stdio.h>

int main() {
    int x = 42;          // 变量 x，值为 42
    int *p = &x;         // p 是一个指针，存放 x 的地址
    
    printf("x 的值: %d\n", x);    // 42
    printf("x 的地址: %p\n", &x); // 0x7ffd...
    printf("p 的值: %p\n", p);    // 和 &x 一样
    printf("p 指向的值: %d\n", *p); // 42（解引用）
    
    *p = 100;            // 通过指针修改 x
    printf("x 的新值: %d\n", x); // 100
    
    return 0;
}
```

### 指针的图解

```
内存地址：0x1000  0x1004  0x1008  0x100C ...
         ┌──────┬──────┬──────┬──────┐
         │  42  │      │ 0x1000│      │
         └──────┴──────┴──────┴──────┘
            ↑                ↑
            x               p (指向 x)
```

## 8.3 ⚡ 文件 I/O 系统调用

### open — 打开文件

```c
#include <fcntl.h>    // open flags
#include <unistd.h>   // read/write/close
#include <stdio.h>
#include <errno.h>

int main() {
    // 打开文件（只读）
    int fd = open("/etc/passwd", O_RDONLY);
    if (fd == -1) {
        perror("打开失败");  // 打印错误信息
        return 1;
    }
    printf("文件描述符: %d\n", fd);  // 通常是 3
    
    close(fd);
    return 0;
}
```

**open 的 flag**：
```c
O_RDONLY   // 只读
O_WRONLY   // 只写
O_RDWR     // 读写
O_CREAT    // 不存在则创建
O_TRUNC    // 清空文件内容
O_APPEND   // 追加
O_EXCL     // 和 O_CREAT 组合，文件存在则报错

// 组合使用（用 | 连接）
int fd = open("test.txt", O_WRONLY | O_CREAT | O_TRUNC, 0644);
//                                                    ↑ 权限（如果创建）
```

### read — 读取文件

```c
#include <unistd.h>

int main() {
    int fd = open("/etc/passwd", O_RDONLY);
    
    char buffer[1024];          // 缓冲区
    ssize_t bytes_read = read(fd, buffer, sizeof(buffer) - 1);
    
    if (bytes_read > 0) {
        buffer[bytes_read] = '\0';  // 字符串结尾
        printf("读取了 %zd 字节:\n%s\n", bytes_read, buffer);
    }
    
    close(fd);
    return 0;
}
```

**read 的返回值**：
```
> 0  → 实际读取的字节数
= 0  → 文件末尾（EOF）
= -1 → 出错（检查 errno）
```

### write — 写入文件

```c
#include <unistd.h>
#include <string.h>

int main() {
    int fd = open("output.txt", O_WRONLY | O_CREAT | O_TRUNC, 0644);
    
    const char *text = "Hello, Linux!\n";
    ssize_t written = write(fd, text, strlen(text));
    
    if (written == -1) {
        perror("写入失败");
        return 1;
    }
    
    printf("写入了 %zd 字节\n", written);
    close(fd);
    return 0;
}
```

### ⚡ 文件描述符的本质

```c
#include <unistd.h>
#include <fcntl.h>

int main() {
    // 标准文件描述符：
    // 0 = stdin（键盘）
    // 1 = stdout（屏幕）
    // 2 = stderr（屏幕）
    
    write(1, "到 stdout\n", 10);    // 等价于 printf
    write(2, "到 stderr\n", 10);   // 错误输出
    
    // 重定向模拟
    int fd = open("log.txt", O_WRONLY | O_CREAT, 0644);
    dup2(fd, 1);                    // 将 stdout 重定向到文件
    printf("这行写到文件而不是屏幕\n");
    
    close(fd);
    return 0;
}
```

**dup2 的原理**：
```
重定向前：
    fd 1 → 屏幕
    fd 3 → 文件

调用 dup2(3, 1) 后：
    fd 1 → 文件 (原来屏幕的引用被关闭)
    fd 3 → 文件 (fd 1 和 fd 3 指向同一个打开的文件)
```

### lseek — 移动文件读写位置

```c
#include <unistd.h>
#include <fcntl.h>

int main() {
    int fd = open("test.txt", O_RDONLY);
    
    char buf[10];
    read(fd, buf, 5);        // 读取前 5 个字节
    
    lseek(fd, 0, SEEK_SET);  // 回到文件开头
    read(fd, buf, 5);        // 又从头读取
    
    lseek(fd, -5, SEEK_END); // 跳到文件末尾前 5 字节
    read(fd, buf, 5);        // 读取最后 5 字节
    
    off_t pos = lseek(fd, 0, SEEK_CUR);  // 当前位置
    off_t size = lseek(fd, 0, SEEK_END); // 文件大小
    
    close(fd);
    return 0;
}
```

## 8.4 ⚡ 进程管理系统调用

### fork — 创建进程

```c
#include <unistd.h>
#include <stdio.h>

int main() {
    pid_t pid = fork();
    
    if (pid == -1) {
        perror("fork 失败");
        return 1;
    }
    
    if (pid == 0) {
        // 子进程
        printf("子进程: PID=%d, 父进程 PPID=%d\n", getpid(), getppid());
        
        for (int i = 0; i < 3; i++) {
            printf("子进程: 第 %d 秒\n", i);
            sleep(1);
        }
    } else {
        // 父进程
        printf("父进程: PID=%d, 子进程 PID=%d\n", getpid(), pid);
        
        for (int i = 0; i < 3; i++) {
            printf("父进程: 第 %d 秒\n", i);
            sleep(1);
        }
    }
    
    // 父和子都会执行这里（各自运行）
    printf("进程 %d 结束\n", getpid());
    return 0;
}
```

**fork 后的内存**：
```
fork 之前：父进程有变量 x = 42
    ↓
fork 之后：
父进程：x = 42（独立副本）
子进程：x = 42（独立副本，修改互不影响）

这就是"写时复制 (Copy on Write)"技术：
    - fork 时并不复制全部内存
    - 只有当父或子修改内存时，才复制被修改的页面
    - 节省了大量内存和时间
```

### exec — 执行新程序

```c
#include <unistd.h>
#include <stdio.h>
#include <sys/wait.h>

int main() {
    pid_t pid = fork();
    
    if (pid == 0) {
        // 子进程：把自己变成 ls
        printf("子进程: 即将执行 ls\n");
        
        execlp("ls", "ls", "-la", NULL);
        
        // 如果 exec 成功，以下代码不会执行
        perror("exec 失败");
        return 1;
    }
    
    // 父进程等待子进程结束
    int status;
    wait(&status);
    
    if (WIFEXITED(status)) {
        printf("子进程正常退出，退出码: %d\n", WEXITSTATUS(status));
    }
    
    return 0;
}
```

**exec 系列函数**：

| 函数 | 说明 |
|---|---|
| execl | 参数列表（以 NULL 结尾） |
| execv | 参数数组 |
| execlp | 参数列表 + 在 PATH 中搜索 |
| execvp | 参数数组 + 在 PATH 中搜索 |
| execle | 参数列表 + 自定义环境变量 |

```c
// 不同形式的 exec
execl("/bin/ls", "ls", "-la", NULL);
execv("/bin/ls", (char*[]){"ls", "-la", NULL, NULL});
execlp("ls", "ls", "-la", NULL);       // 自动在 PATH 找
execvp("ls", (char*[]){"ls", "-la", NULL, NULL});  // 自动在 PATH 找
```

### wait — 等待子进程

```c
#include <sys/wait.h>

int main() {
    pid_t pid = fork();
    
    if (pid == 0) {
        sleep(2);
        return 42;  // 子进程退出码 42
    }
    
    // 父进程
    int status;
    pid_t child_pid = wait(&status);  // 阻塞，直到子进程结束
    
    if (WIFEXITED(status)) {
        printf("子进程 %d 退出，退出码: %d\n", 
               child_pid, WEXITSTATUS(status));
        // 退出码 42
    }
    
    if (WIFSIGNALED(status)) {
        printf("子进程被信号 %d 杀死\n", WTERMSIG(status));
    }
    
    return 0;
}
```

### ⚡ Shell 的完整实现（简化）

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/wait.h>

int main() {
    char cmd[256];
    
    while (1) {
        printf("mysh$ ");
        fflush(stdout);
        
        if (fgets(cmd, sizeof(cmd), stdin) == NULL) {
            break;  // Ctrl+D
        }
        
        // 去掉换行符
        cmd[strcspn(cmd, "\n")] = '\0';
        
        if (strcmp(cmd, "exit") == 0) {
            break;
        }
        
        pid_t pid = fork();
        
        if (pid == 0) {
            // 子进程
            char *args[] = {"/bin/sh", "-c", cmd, NULL};
            execvp(args[0], args);
            perror("执行失败");
            exit(1);
        } else {
            // 父进程等待
            wait(NULL);
        }
    }
    
    return 0;
}
```

## 8.5 内存管理系统调用

### brk/sbrk — 调整堆

```c
#include <unistd.h>
#include <stdio.h>

int main() {
    // 获取当前堆结束位置
    void *current_break = sbrk(0);
    printf("当前堆顶: %p\n", current_break);
    
    // 分配 1024 字节
    sbrk(1024);
    
    void *new_break = sbrk(0);
    printf("分配后堆顶: %p\n", new_break);
    printf("分配了 %ld 字节\n", (long)(new_break - current_break));
    
    // 实际上你不会直接使用 sbrk
    // 而是用 malloc/free（底层调用 brk/sbrk 或 mmap）
    
    return 0;
}
```

### malloc/free — 内存分配器

```c
#include <stdlib.h>
#include <stdio.h>
#include <string.h>

int main() {
    // 动态分配内存
    int *arr = malloc(10 * sizeof(int));  // 分配 10 个 int
    if (arr == NULL) {
        fprintf(stderr, "分配失败\n");
        return 1;
    }
    
    // 使用
    for (int i = 0; i < 10; i++) {
        arr[i] = i * 10;
    }
    
    // 扩容
    int *new_arr = realloc(arr, 20 * sizeof(int));
    if (new_arr != NULL) {
        arr = new_arr;
    }
    
    // 清零
    memset(arr, 0, 20 * sizeof(int));
    
    // 释放
    free(arr);  // 归还给内存池
    
    // 危险！释放后继续使用（use-after-free）
    // arr[0] = 42;  // 未定义行为！
    
    return 0;
}
```

**malloc 的实现原理**：
```
malloc(100)
    ↓
内存分配器（ptmalloc / jemalloc / tcmalloc）
    ↓
检查空闲链表中有无合适的内存块
    ├── 有：直接返回
    └── 无：调用 brk() 或 mmap() 向内核要更多内存
         ↓
    内核分配物理内存页
         ↓
    返回虚拟地址
```

### mmap — 内存映射

mmap 是一种更灵活的内存分配/文件操作方式。

```c
#include <sys/mman.h>
#include <stdio.h>
#include <string.h>
#include <unistd.h>

int main() {
    size_t size = 4096;  // 1 页
    
    // 分配匿名内存（不映射文件）
    void *addr = mmap(NULL, size, 
                      PROT_READ | PROT_WRITE,  // 读写权限
                      MAP_PRIVATE | MAP_ANONYMOUS,  // 私有匿名
                      -1, 0);  // 没有文件描述符
    
    if (addr == MAP_FAILED) {
        perror("mmap 失败");
        return 1;
    }
    
    // 像普通内存一样使用
    strcpy(addr, "Hello from mmap!");
    printf("%s\n", (char*)addr);
    
    // 释放
    munmap(addr, size);
    
    return 0;
}
```

**mmap 的两种场景**：

```
场景 1：分配内存（MAP_ANONYMOUS）
    mmap → 内核分配零填充的物理页
    适用：大块内存分配（> 128KB）

场景 2：文件映射
    mmap → 将文件内容映射到内存
    读写内存 = 读写文件（无需 read/write 系统调用）
```

## 8.6 ⚡ 进程间通信（IPC）

### 管道（PIPE）

```c
#include <unistd.h>
#include <stdio.h>
#include <string.h>

int main() {
    int pipefd[2];
    pipe(pipefd);  // 创建管道
                   // pipefd[0] = 读端
                   // pipefd[1] = 写端
    
    pid_t pid = fork();
    
    if (pid == 0) {
        // 子进程：关闭写端，从管道读
        close(pipefd[1]);
        
        char buf[256];
        read(pipefd[0], buf, sizeof(buf));
        printf("子进程收到: %s\n", buf);
        close(pipefd[0]);
        
    } else {
        // 父进程：关闭读端，向管道写
        close(pipefd[0]);
        
        const char *msg = "Hello from parent!";
        write(pipefd[1], msg, strlen(msg) + 1);
        close(pipefd[1]);
        
        wait(NULL);
    }
    
    return 0;
}
```

**管道原理**：
```
父进程                                   子进程
   │                                       │
   ├── pipefd[1]（写）                     │
   ├── 写入 "hello" ──────────────→        │
   │                   (内核缓冲区)          │
   │                                       ├── pipefd[0]（读）
   │                                       ├── 读出 "hello"
```

**命令行的管道等价实现**：
```bash
# Shell 中的 ls | grep txt 等价于：
# 1. 创建管道
# 2. fork 两个子进程
# 3. 子进程 1：ls 的 stdout → 管道写端
# 4. 子进程 2：grep 的 stdin ← 管道读端
```

### 信号（Signal）

```c
#include <signal.h>
#include <stdio.h>
#include <unistd.h>

// 信号处理函数
volatile sig_atomic_t interrupted = 0;

void handler(int sig) {
    interrupted = 1;
    write(1, "!\n", 2);  // 在信号处理器中只能调用 async-signal-safe 函数
}

int main() {
    // 注册 SIGINT 处理器
    struct sigaction sa;
    sa.sa_handler = handler;
    sigemptyset(&sa.sa_mask);
    sa.sa_flags = 0;
    sigaction(SIGINT, &sa, NULL);
    // 旧方式：signal(SIGINT, handler);
    
    printf("运行中... 按 Ctrl+C 试试\n");
    
    while (!interrupted) {
        write(1, ".", 1);
        sleep(1);
    }
    
    printf("\n优雅退出\n");
    return 0;
}
```

### 共享内存

```c
#include <sys/mman.h>
#include <unistd.h>
#include <stdio.h>
#include <string.h>
#include <sys/wait.h>

int main() {
    // 创建共享内存映射
    size_t size = 4096;
    int *shared = mmap(NULL, size,
                       PROT_READ | PROT_WRITE,
                       MAP_SHARED | MAP_ANONYMOUS,
                       -1, 0);
    
    *shared = 0;  // 初始值
    
    pid_t pid = fork();
    
    if (pid == 0) {
        // 子进程
        *shared = 42;  // 修改共享变量
        printf("子进程设置共享值为: %d\n", *shared);
    } else {
        // 父进程
        sleep(1);  // 等子进程修改
        printf("父进程看到共享值: %d\n", *shared);  // 42！
        
        wait(NULL);
        munmap(shared, size);
    }
    
    return 0;
}
```

## 8.7 ⚡ 虚拟文件系统（VFS）

VFS 是 Linux 文件系统的核心抽象。

```
用户程序
    │
    ├── write(fd, ...)
    │
    ↓
VFS（虚拟文件系统）
    │  统一的接口：write, read, open, close...
    │
    ├── ext4（硬盘文件系统）
    ├── ntfs（Windows 兼容）
    ├── proc（进程信息）
    ├── sysfs（系统信息）
    ├── tmpfs（内存中的文件系统）
    ├── nfs（网络文件系统）
    └── ...
```

VFS 的核心数据结构：

```c
// 超级块 — 文件系统的元信息
struct super_block {
    // ...
};

// inode — 文件的元数据（权限、大小、位置）
struct inode {
    umode_t i_mode;    // 文件类型和权限
    uid_t i_uid;       // 所有者
    off_t i_size;      // 文件大小
    struct timespec i_atime;  // 访问时间
    // ...
};

// dentry — 目录项（路径名和 inode 的映射）
struct dentry {
    struct inode *d_inode;   // 对应的 inode
    struct dentry *d_parent; // 父目录
    // ...
};

// file — 打开的文件（进程视角）
struct file {
    struct dentry *f_dentry; // 对应的 dentry
    loff_t f_pos;            // 当前读写位置
    struct file_operations *f_op;  // 操作函数表
    // ...
};
```

### 从 open 到硬盘的完整流程

```
open("/etc/passwd", O_RDONLY)
    ↓
VFS 查找路径：/etc/passwd
    ↓
逐级查找 dentry 缓存：
    "/" → dentry
    "etc" → dentry  
    "passwd" → dentry → inode
    ↓
创建 file 结构体，关联 dentry/inode
    ↓
调用文件系统的 open 函数
    ↓
返回文件描述符 (fd = 3)
```

## 8.8 动态链接库

### 静态链接 vs 动态链接

```
静态链接：
main.c → 编译 → main.o
lib.c  → 编译 → lib.o
链接：main.o + lib.o → a.out（单个可执行文件，体积大）

动态链接：
main.c → 编译 → main.o
链接：main.o + lib.so（运行时加载，体积小，共享内存）
```

### 创建和使用动态库

```c
// libmath.c
int add(int a, int b) { return a + b; }
int sub(int a, int b) { return a - b; }
```

```bash
# 编译成动态库
gcc -shared -fPIC -o libmath.so libmath.c
# -shared 生成共享库
# -fPIC 位置无关代码

# 使用动态库
# main.c
#include <stdio.h>
int add(int, int);

int main() {
    printf("3 + 5 = %d\n", add(3, 5));
    return 0;
}

# 编译
gcc main.c -L. -lmath -o program
# -L. 在当前目录搜索库
# -lmath 链接 libmath.so

# 运行前需要告诉系统库的位置
export LD_LIBRARY_PATH=./
./program
```

### ⚡ 动态链接的原理

```
程序加载时：
    动态链接器（ld-linux.so）读取 ELF 文件的 .dynamic 段
        ↓
    找到需要的共享库列表
        ↓
    在标准路径中搜索：/lib, /usr/lib, /etc/ld.so.conf
        ↓
    加载共享库到内存
        ↓
    重定位：将库中的函数地址填入 GOT（全局偏移表）
        ↓
    程序开始执行
```

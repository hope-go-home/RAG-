## 完整代码

```python
"""
chat_app.py - 完整 AI 聊天应用（RAG + Agent）
运行: streamlit run chat_app.py
"""

import streamlit as st
import tempfile
import os

from langchain_openai import ChatOpenAI, OpenAIEmbeddings
from langchain_chroma import Chroma
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_community.document_loaders import (
    TextLoader, PyPDFLoader, Docx2txtLoader
)
from langchain_core.prompts import ChatPromptTemplate, MessagesPlaceholder
from langchain_core.output_parsers import StrOutputParser
from langchain_core.runnables import RunnablePassthrough
from langchain_core.messages import HumanMessage, AIMessage
from langchain_core.tools import tool
from langchain.agents import create_openai_tools_agent, AgentExecutor

# ============================================================
# 一、页面配置
# ============================================================
st.set_page_config(page_title="AI 助手", page_icon="🤖", layout="wide")

# ============================================================
# 二、初始化
# ============================================================
def init_state():
    st.session_state.setdefault("messages", [])
    st.session_state.setdefault("vectorstore", None)
    st.session_state.setdefault("rag_chain", None)
    st.session_state.setdefault("mode", "chat")

init_state()

# ============================================================
# 三、LLM 初始化
# ============================================================
@st.cache_resource
def get_llm():
    return ChatOpenAI(
        model="gpt-4o-mini",
        temperature=0.7,
        streaming=True
    )

@st.cache_resource
def get_embeddings():
    return OpenAIEmbeddings(model="text-embedding-3-small")

llm = get_llm()
embeddings = get_embeddings()

# ============================================================
# 四、文档处理（RAG）
# ============================================================
def process_document(uploaded_file):
    """处理上传的文档，创建向量库"""
    # 保存临时文件
    with tempfile.NamedTemporaryFile(
        delete=False,
        suffix=os.path.splitext(uploaded_file.name)[1]
    ) as tmp:
        tmp.write(uploaded_file.getvalue())
        tmp_path = tmp.name

    # 根据扩展名选择加载器
    ext = os.path.splitext(uploaded_file.name)[1].lower()
    if ext == ".txt":
        loader = TextLoader(tmp_path, encoding="utf-8")
    elif ext == ".pdf":
        loader = PyPDFLoader(tmp_path)
    elif ext == ".docx":
        loader = Docx2txtLoader(tmp_path)
    else:
        st.error(f"不支持的文件格式: {ext}")
        return None

    # 加载文档
    docs = loader.load()

    # 分割
    splitter = RecursiveCharacterTextSplitter(
        chunk_size=500,
        chunk_overlap=50
    )
    chunks = splitter.split_documents(docs)

    # 创建向量库
    vectorstore = Chroma.from_documents(
        documents=chunks,
        embedding=embeddings,
        persist_directory="./chroma_db"
    )

    # 清理临时文件
    os.unlink(tmp_path)

    return vectorstore

def build_rag_chain(vectorstore):
    """构建 RAG 链"""
    retriever = vectorstore.as_retriever(
        search_kwargs={"k": 3}
    )

    prompt = ChatPromptTemplate([
        ("system", """你是一个知识问答助手。
请基于提供的资料回答问题。
如果你从资料中找不到答案，明确说"资料中没有相关信息"。
不要编造答案。

参考资料：
{context}"""),
        MessagesPlaceholder("history"),
        ("human", "{question}")
    ])

    def format_docs(docs):
        return "\n\n---\n\n".join(doc.page_content for doc in docs)

    rag_chain = (
        {
            "context": retriever | format_docs,
            "history": lambda x: st.session_state.messages[-10:],
            "question": RunnablePassthrough()
        }
        | prompt
        | llm
        | StrOutputParser()
    )

    return rag_chain

# ============================================================
# 五、Agent 工具
# ============================================================
@tool
def web_search(query: str) -> str:
    """搜索网络获取最新信息。输入：搜索关键词"""
    from langchain_community.tools import DuckDuckGoSearchRun
    search = DuckDuckGoSearchRun()
    try:
        return search.invoke(query)
    except:
        return "搜索失败，请稍后重试"

@tool
def calculate(expression: str) -> str:
    """执行数学计算。输入：数学表达式，如 '2**10' 或 '123*456'"""
    try:
        return str(eval(expression))
    except Exception as e:
        return f"计算错误: {e}"

tools = [web_search, calculate]

def build_agent():
    """构建 Agent"""
    prompt = ChatPromptTemplate([
        ("system", "你是全能助手。可以用工具搜索和计算。回答要简洁准确。"),
        MessagesPlaceholder("history"),
        ("human", "{input}"),
        ("placeholder", "{agent_scratchpad}")
    ])

    agent = create_openai_tools_agent(llm, tools, prompt)

    agent_executor = AgentExecutor(
        agent=agent,
        tools=tools,
        verbose=True,
        max_iterations=5,
        handle_parsing_errors=True
    )

    return agent_executor

# ============================================================
# 六、主界面
# ============================================================
def main():
    # 侧边栏
    with st.sidebar:
        st.title("🤖 AI 助手")

        # 模式切换
        st.session_state.mode = st.radio(
            "选择模式",
            ["💬 聊天", "📚 文档问答"],
            index=0 if st.session_state.mode == "chat" else 1
        )

        st.divider()

        # RAG 模式：上传文档
        if "文档问答" in st.session_state.mode:
            st.subheader("📄 上传知识文档")
            uploaded_file = st.file_uploader(
                "支持 TXT / PDF / DOCX",
                type=["txt", "pdf", "docx"]
            )

            if uploaded_file:
                with st.spinner("正在处理文档..."):
                    vectorstore = process_document(uploaded_file)
                    if vectorstore:
                        st.session_state.vectorstore = vectorstore
                        st.session_state.rag_chain = build_rag_chain(vectorstore)
                        st.success(f"✅ 已处理: {uploaded_file.name}")
                    else:
                        st.error("文档处理失败")

            if st.session_state.vectorstore:
                if st.button("🗑️ 清除文档"):
                    st.session_state.vectorstore = None
                    st.session_state.rag_chain = None
                    import shutil
                    if os.path.exists("./chroma_db"):
                        shutil.rmtree("./chroma_db")
                    st.rerun()

        # 控制
        st.divider()
        if st.button("🗑️ 清除对话"):
            st.session_state.messages = []
            st.rerun()

    # 主区域
    if "💬 聊天" in st.session_state.mode:
        st.title("💬 智能聊天")
        st.caption("支持网络搜索和数学计算")
        mode_type = "chat"
    else:
        st.title("📚 文档知识问答")
        if st.session_state.vectorstore:
            st.caption("已加载知识库，可以基于文档提问")
            mode_type = "rag"
        else:
            st.info("请在左侧上传知识文档")
            mode_type = "rag_no_doc"

    # 显示历史消息
    for msg in st.session_state.messages:
        role = "user" if isinstance(msg, HumanMessage) else "assistant"
        with st.chat_message(role):
            st.write(msg.content)

    # 输入
    if prompt := st.chat_input("请输入你的问题..."):
        # 显示用户消息
        st.session_state.messages.append(HumanMessage(content=prompt))
        with st.chat_message("user"):
            st.write(prompt)

        # AI 回复
        with st.chat_message("assistant"):
            with st.spinner("思考中..."):
                try:
                    if mode_type == "chat":
                        agent = build_agent()
                        result = agent.invoke({
                            "input": prompt,
                            "history": st.session_state.messages[-10:-1]
                        })
                        response = result["output"]
                    elif mode_type == "rag":
                        response = st.session_state.rag_chain.invoke(prompt)
                    else:
                        response = "请先在左侧上传文档"

                    st.write(response)
                    st.session_state.messages.append(
                        AIMessage(content=response)
                    )

                except Exception as e:
                    st.error(f"出错了: {e}")
                    st.session_state.messages.append(
                        AIMessage(content=f"抱歉，出错了: {str(e)}")
                    )

if __name__ == "__main__":
    main()
```
